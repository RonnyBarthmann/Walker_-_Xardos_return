MU  HI LA FEE = VII - Feuersturm
MEK    LA NOP =  II - Feuerkugel
ME     LA NOP =   I - Feuerkügelchen
MO     LE REP =   V - Erde oder Pflanzen bändigen
MO     LA REP =   V - Feuerbändigen
MO     LI VOK =   V - Heilung
MO     LI REP =   V - Innere Wunden heilen
MO     LO VOK =   V - Äußere Wunden schließen
MO     LY NOP =   V - Manakugel schießen, die Gegner lehmt.

----------------------------------------------------------------

Stufe:  MU  = VII (7)
        MA  = VI  (6)
        MO  = V   (5)
        MIK = IV  (4)
        MI  = III (3)
  (mäk) MEK = II  (2)
        ME  = I   (1)

Zusatz: HI = Ohne Bereichslimit

Klasse: LA = Feuer
        LO = Wasser
        LU = Blitz
        LE = Natur
        LI = Licht
   (lü) LY = Schatten/Nekromanie

Zauber:    FEE = Flut (bis Stufe 2 auch Spruchlos)
        LA FEE = Feuersturm
        LO FEE = Welle
        LU FEE = vom Wirker ausgehende Blitzwolke (wie eine Plasmakugel)
        LE FEE = Erdbeben
        LI FEE = Licht (LI FEE = Lichtkegel / HI LI FEE = in alle Richtungen)
        LY FEE = Dunkelheit (umschließt das Opfer)
        
           NOP = Kugel (bis Stufe 4 auch Spruchlos)
        LA NOP = Feuerball   (kann ab Stuffe 2 verdichtet werden, wird dann schwarz)
        LO NOP = Eiskristall (kann ab Stuffe 2 verdichtet werden, wird dann dunkelblau)
        LU NOP = kleiner elektrischer Schlag (wie ein Taser)
        LE NOP = Steine (können in hoher abfolge geschossen werden)
        LI NOP = kleines Licht (wie eine Kerze, heilt bei Berührung)
        LY NOP = Mana kugel, entzieht Leben (lehmt bei Berührung temporär, ab Stuffe 6 permanent)

           REP = Steuern von Energie / "Bändigen" (mindestens Stufe 3 erforderlich)
        LA REP = Feuerbändigen
        LO REP = Wasserbändigen (und Eis)

        LE REP = Erdbändigen / Ranken, Pflanzen und Wurzeln bändigen (ab Stufe 5 auch Metalle/Eisen)
        LI REP = innere wunden Heilen (auch Statuseffekte wie Überlkeit und Vergiftung)
        LY REP = Geister Befehligen

           VOK = Heilen (allgemein, aber mit Klassen-spezifischen vorteilen)
        LA VOK = Heilt, besonders effektiv bei Verbrennungen
        LO VOK = Heilt, besonders effektiv bei oberflächlichen Verletzungen
        LU VOK = Heilt, besonders effektiv bei psychischer Beeinträchtigungen
        LE VOK = Heilt, besonders effektiv bei Vergiftungen / Statusbeinträchtigungen.
        LI VOK = Heilt effektiver als die anderen Klassen, dafür keine Spezialisierung
        LY VOK = Heilt bestehende Flüche und befreit Körper vor Parasiten (besessene Körper)

           NIP = AoE (ab Stufe 4)
        LA NIP = Feuer (aus dem Boden oder als Regen)
        LO NIP = Regen, aus dem Boden sickerndes Wasser, Nebel
        LU NIP = Blitzwolke
        LE NIP = Aus dem Boden schießende Ranken und herumwirbelnde Blütensamen
        LI NIP = AoE Heilung (wie LI VOK)
        LY NIP = Dunkelheit, Verwirrung

        LU NEP = Gerichteter Blitz
        LE NEP = Ranke die aus dem Boden schnellt und dann nach vorne Schießt
